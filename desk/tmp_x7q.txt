const DESK_AI_ACCOUNT = 'pausespace1803@gmail.com'; // 排程 AI 用這個帳號讀寫收件夾
function deskRoot_() {
  const P = PropertiesService.getScriptProperties();
  const id = P.getProperty('DESK_DIR');
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  const f = DriveApp.createFolder('08_島主櫃台收件（AI 整理用）');
  try { f.addEditor(DESK_AI_ACCOUNT); } catch (e) {}
  P.setProperty('DESK_DIR', f.getId());
  return f;
}
function deskJson_(folder, name) { const it = folder.getFilesByName(name); if (!it.hasNext()) return null; try { return JSON.parse(it.next().getBlob().getDataAsString('UTF-8')); } catch (e) { return null; } }
function deskPut_(folder, name, obj) {
  const txt = typeof obj === 'string' ? obj : JSON.stringify(obj, null, 2);
  const it = folder.getFilesByName(name);
  if (it.hasNext()) { it.next().setContent(txt); return; }
  folder.createFile(name, txt, name.slice(-5) === '.json' ? 'application/json' : MimeType.PLAIN_TEXT);
}
function deskCase_(id) { const f = DriveApp.getFolderById(id); if (!deskJson_(f, '案件.json')) throw new Error('找不到這個案件'); return f; }

function deskBookings(key) {
  checkKey_(key);
  const now = new Date(), a = new Date(now.getTime() - 10 * 864e5), b = new Date(now.getTime() + 2 * 864e5);
  const out = [];
  calendar_().getEvents(a, b).forEach(function (e) {
    const t = e.getTitle() || ''; if (t.indexOf(CONFIG.TAG) !== 0 || /已取消/.test(t)) return;
    const p = parseEvent_(e); if (!p.name) return;
    out.push({name: p.name, phone: phoneNorm_(p.phone), service: p.service, date: p.date, label: p.label + ' ' + p.start, code: p.code});
  });
  return {types: DESK_TYPES, items: out.sort(function (x, y) { return x.date < y.date ? 1 : -1; })};
}

function deskCreate(key, meta) {
  checkKey_(key);
  meta = meta || {};
  const type = DESK_TYPES[meta.type]; if (!type || type.soon) throw new Error('這個類型還沒開放');
  const name = String(meta.name || '').trim().slice(0, 30), phone = phoneNorm_(meta.phone || '');
  if (!name) throw new Error('請選客人或填名字');
  const date = /^\d{4}-\d{2}-\d{2}$/.test(meta.date || '') ? meta.date : Utilities.formatDate(new Date(), CONFIG.TZ, 'yyyy-MM-dd');
  const f = deskRoot_().createFolder(date.replace(/-/g, '') + '_' + (phone || '無手機') + '_' + name + '_' + type.name);
  deskPut_(f, '案件.json', {id: f.getId(), type: meta.type, typeName: type.name, name: name, phone: phone, date: date,
    code: String(meta.code || ''), service: String(meta.service || ''), note: String(meta.note || '').slice(0, 1000),
    created: Utilities.formatDate(new Date(), CONFIG.TZ, 'yyyy-MM-dd HH:mm')});
  return {id: f.getId()};
}
// 收材料：文字直接存 txt；照片用 base64 傳上來
function deskText(key, id, slot, text) {
  checkKey_(key);
  const f = deskCase_(id), t = String(text || '').trim(); if (!t) return {ok: false};
  deskPut_(f, String(slot || 'text').replace(/\W/g, '') + '.txt', t);
  return {ok: true, chars: t.length};
}
function deskFile(key, id, slot, name, mime, b64) {
  checkKey_(key);
  const f = deskCase_(id);
  const s = String(slot || 'file').replace(/\W/g, ''), n = f.getFiles(); let c = 1;
  while (n.hasNext()) { if (n.next().getName().indexOf(s + '_') === 0) c++; }
  const ext = (String(name || '').match(/\.(\w{2,5})$/) || [, /png/.test(mime) ? 'png' : /text/.test(mime) ? 'txt' : 'jpg'])[1];
  f.createFile(Utilities.newBlob(Utilities.base64Decode(String(b64 || '')), mime || 'image/jpeg', s + '_' + c + '.' + ext));
  return {ok: true};
}
function deskTxt_(f, name) { const it = f.getFilesByName(name); return it.hasNext() ? it.next().getBlob().getDataAsString('UTF-8') : ''; }
function deskStatus_(f) {
  if (deskJson_(f, '已送出.json')) return 'sent';
  if (deskJson_(f, '草稿.json')) return 'draft';
  return 'wait';
}
function deskList(key) {
  checkKey_(key);
  const out = [], it = deskRoot_().getFolders();
  while (it.hasNext()) {
    const f = it.next(), meta = deskJson_(f, '案件.json'); if (!meta) continue;
    const files = []; const fi = f.getFiles(); while (fi.hasNext()) { const x = fi.next(); files.push(x.getName()); }
    let st = deskStatus_(f), before = null, after = null;
    if (meta.type === 'skin') { try { before = JSON.parse(deskTxt_(f, 'before.txt') || 'null'); } catch (e) {} try { after = JSON.parse(deskTxt_(f, 'after.txt') || 'null'); } catch (e) {} if (st === 'wait' && !after) st = 'pre'; }
    out.push({id: f.getId(), meta: meta, status: st, files: files.filter(function (n) { return !/\.json$/.test(n); }), before: before, after: after,
      draft: st === 'draft' ? deskJson_(f, '草稿.json') : null, sent: st === 'sent' ? deskJson_(f, '已送出.json') : null, url: f.getUrl()});
  }
  return out.sort(function (a, b) { return a.meta.created < b.meta.created ? 1 : -1; }).slice(0, 40);
}

}
