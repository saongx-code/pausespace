## 步驟 2：在「程式碼.gs」最下方貼上

```js
/* ===== 靈感箱／語音日記（GitHub admin/inbox.html 送進來） ===== */
var INBOX_SHEET_ID = '16G-8bv-Fy6lzP7fUa2dky6_YIN1PhIrVTCEXdkKvz2g';
var INBOX_FOLDER_ID = '15dehDhNX1Ewp195Xjuq3ePRjxqIWSFCo';

function inboxPost_(b) {
  var out;
  try {
    if (!b.key || b.key !== ADMIN_KEY) throw new Error('鑰匙不正確');
    out = {ok: true, result: inboxSave_(b)};
  } catch (err) {
    out = {error: String((err && err.message) || err)};
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function inboxFolder_() {
  var parent = DriveApp.getFolderById(INBOX_FOLDER_ID);
  var it = parent.getFoldersByName('靈感箱素材');
  return it.hasNext() ? it.next() : parent.createFolder('靈感箱素材');
}

function inboxFile_(f, prefix) {
  if (!f || !f.data) return '';
  var blob = Utilities.newBlob(Utilities.base64Decode(f.data), f.type || 'application/octet-stream', prefix + '_' + (f.name || 'file'));
  return inboxFolder_().createFile(blob).getUrl();
}

function inboxCell_(s) {           // 避免文字開頭是 = + - @ 被當成公式
  s = String(s == null ? '' : s);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}

function inboxSave_(b) {
  var ss = SpreadsheetApp.openById(INBOX_SHEET_ID);
  var now = new Date();
  var day = Utilities.formatDate(now, 'Asia/Taipei', 'yyyy-MM-dd HH:mm');
  var stamp = Utilities.formatDate(now, 'Asia/Taipei', 'yyyyMMdd_HHmmss');
  var audio = inboxFile_(b.audio, stamp + '_錄音');

  if (b.mode === 'diary') {
    // 日期｜標題｜故事內容（語音轉文字）｜情緒／主題標籤｜可以延伸的議題｜原始錄音連結｜已用在｜備註
    ss.getSheetByName('日記').appendRow([
      day, inboxCell_(b.title), inboxCell_(b.text), '', '', audio, '',
      b.text ? '' : '待轉文字'
    ]);
    return {sheet: '日記'};
  }

  var imgs = (b.images || []).map(function (f, i) { return inboxFile_(f, stamp + '_圖' + (i + 1)); }).filter(String);
  var extra = [];
  if (b.text) extra.push('語音文字：' + b.text);
  if (audio) extra.push('錄音：' + audio);
  // 日期｜類型｜連結或檔案｜喜歡的地方｜可以借鏡的手法｜帳號／來源｜圖片截圖｜備註
  ss.getSheetByName('靈感對標').appendRow([
    day, b.kind || '', inboxCell_(b.url || imgs[0] || ''), inboxCell_(b.note), '',
    inboxCell_(b.source), imgs.join('\n'), inboxCell_(extra.join('\n'))
  ]);
  return {sheet: '靈感對標'};
}
```
