const INSP_DAILY = '1H0Eqyqv0b_0VyHmil4k2zx29a1qh2PUN';
const INSP_USED = '1GQXHCywAm8Q-s0IncmXhxGthqBrmlIOw';
const INSP_USAGES = ['Threads', '輪播', '短影音', '限動'];
const INSP_ROOT = '15AU5P-K3CQresOqh7XSdwSkPKn2QkreM';
const INSP_FB_KINDS = ['喜歡', '不喜歡', '想要更多', '不需要'];
const INSP_USAGE_DIR = {'Threads': 'Threads', '輪播': '輪播貼文', '短影音': '短影音', '限動': '限動'};

function inspLatest_() {
  const it = DriveApp.getFolderById(INSP_DAILY).getFiles();
  let best = null, bestName = '';
  while (it.hasNext()) {
    const f = it.next(), n = f.getName();
    if (/^\d{4}-\d{2}-\d{2}\.json$/.test(n) && n > bestName) { best = f; bestName = n; }
  }
  return best;
}
function inspFile_(date) {
  const it = DriveApp.getFolderById(INSP_DAILY).getFilesByName(date + '.json');
  return it.hasNext() ? it.next() : null;
}
function inspRead_(f) {
  const cache = CacheService.getScriptCache(), ck = 'insp_' + f.getId() + '_' + f.getLastUpdated().getTime();
  let s = cache.get(ck);
  if (!s) { s = f.getBlob().getDataAsString('UTF-8'); if (s.length < 90000) cache.put(ck, s, 21600); }
  return JSON.parse(s);
}
function inspireToday(key) {
  checkKey_(key);
  const today = Utilities.formatDate(new Date(), CONFIG.TZ, 'yyyy-MM-dd');
  const f = inspFile_(today) || inspLatest_();
  if (!f) return {ok: false, today: today};
  const d = inspRead_(f);
  const saved = JSON.parse(PropertiesService.getScriptProperties().getProperty('INSP_SAVED') || '{}');
  const fb = JSON.parse(PropertiesService.getScriptProperties().getProperty('INSP_FBSTATE') || '{}');
  return {ok: true, today: today, data: d, saved: saved[d.date] || {}, fb: fb[d.date] || {}};
}
function inspSub_(parent, name) {
  const it = parent.getFoldersByName(name);
  return it.hasNext() ? it.next() : parent.createFolder(name);
}
function inspireSave(key, date, id, usage) {
  checkKey_(key);
  if (INSP_USAGES.indexOf(usage) < 0) throw new Error('用途不對');
  const f = inspFile_(String(date));
  if (!f) throw new Error('找不到這天的靈感');
  const d = inspRead_(f), x = (d.items || []).filter(function (i) { return i.id === id; })[0];
  if (!x) throw new Error('找不到這則');
  const proj = (x.projects || [])[0] || '未分類';
  const lines = ['【' + (x.projects || []).join('｜') + '】' + x.title, '日期：' + d.date + '　用途：' + usage, '', '為什麼可以用：' + x.why, '', '開頭：' + (x.hook || '')];
  if (x.threads) lines.push('', '— Threads 文案 —', x.threads);
  if ((x.carousel || []).length) lines.push('', '— 輪播 —', x.carousel.join('\n'));
  if ((x.reel || []).length) lines.push('', '— 短影音分鏡 —', x.reel.join('\n'));
  lines.push('', '法規提醒：' + (x.compliance || ''), '來源：' + ((x.source || {}).title || '') + ' ' + ((x.source || {}).url || ''));
  const dir = inspSub_(DriveApp.getFolderById(INSP_USED), INSP_USAGE_DIR[usage]);
  const file = dir.createFile('【' + proj + '】' + d.date + '｜' + x.title + '.txt', lines.join('\n'), MimeType.PLAIN_TEXT);
  const props = PropertiesService.getScriptProperties();
  const saved = JSON.parse(props.getProperty('INSP_SAVED') || '{}');
  const keep = {}; Object.keys(saved).sort().slice(-30).forEach(function (k) { keep[k] = saved[k]; });
  keep[d.date] = keep[d.date] || {}; keep[d.date][id + '|' + usage] = file.getUrl();
  props.setProperty('INSP_SAVED', JSON.stringify(keep));
  return {ok: true, url: file.getUrl(), saved: keep[d.date]};
}

function inspFbFile_() {
  const root = DriveApp.getFolderById(INSP_ROOT), it = root.getFilesByName('feedback.json');
  return it.hasNext() ? it.next() : root.createFile('feedback.json', '[]', 'application/json');
}
function inspireFeedback(key, date, id, kind, note) {
  checkKey_(key);
  if (INSP_FB_KINDS.indexOf(kind) < 0) throw new Error('回饋種類不對');
  const lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    const f = inspFile_(String(date)), d = f ? inspRead_(f) : {items: []};
    const x = (d.items || []).filter(function (i) { return i.id === id; })[0] || {};
    const ff = inspFbFile_(); let list = [];
    try { list = JSON.parse(ff.getBlob().getDataAsString('UTF-8') || '[]'); } catch (e) { list = []; }
    list.push({date: String(date), id: id, title: x.title || '', kind: x.kind || '', projects: x.projects || [], usage: x.usage || [],
      feedback: kind, note: String(note || '').slice(0, 300), at: Utilities.formatDate(new Date(), CONFIG.TZ, "yyyy-MM-dd'T'HH:mm")});
    list = list.slice(-300);
    ff.setContent(JSON.stringify(list));
    const props = PropertiesService.getScriptProperties();
    const st = JSON.parse(props.getProperty('INSP_FBSTATE') || '{}');
    const keep = {}; Object.keys(st).sort().slice(-20).forEach(function (k) { keep[k] = st[k]; });
    keep[date] = keep[date] || {}; keep[date][id] = kind;
    props.setProperty('INSP_FBSTATE', JSON.stringify(keep));
    return {ok: true, fb: keep[date]};
  } finally { lock.releaseLock(); }
}
